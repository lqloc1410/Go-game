cmake_minimum_required(VERSION 3.16)
project(GoGame LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SFML
    GIT_REPOSITORY https://github.com/SFML/SFML.git
    GIT_TAG 2.6.1)
set(SFML_USE_SYSTEM_DEPS OFF CACHE BOOL "" FORCE)

if(SFML_USE_SYSTEM_DEPS)
    list(APPEND CMAKE_IGNORE_PATH
        "${CMAKE_BINARY_DIR}/_deps/sfml-src/extlibs/headers"
        "${CMAKE_BINARY_DIR}/_deps/sfml-src/extlibs/libs-mingw/x64"
        "${CMAKE_BINARY_DIR}/_deps/sfml-src/extlibs/libs-mingw/x86"
    )

    set(SFML_DEPENDENCY_CACHE_VARS
        FLAC_INCLUDE_DIR FLAC_LIBRARY
        FREETYPE_INCLUDE_DIR_ft2build FREETYPE_INCLUDE_DIR_freetype2 FREETYPE_LIBRARY
        OGG_INCLUDE_DIR OGG_LIBRARY
        OPENAL_INCLUDE_DIR OPENAL_LIBRARY
        VORBIS_INCLUDE_DIR VORBIS_LIBRARY VORBISFILE_LIBRARY VORBISENC_LIBRARY
    )

    foreach(SFML_DEPENDENCY_VAR IN LISTS SFML_DEPENDENCY_CACHE_VARS)
        if("${${SFML_DEPENDENCY_VAR}}" MATCHES "extlibs")
            unset(${SFML_DEPENDENCY_VAR} CACHE)
        endif()
    endforeach()
endif()

FetchContent_MakeAvailable(SFML)

option(GO_NATIVE "Compile with -march=native (fast, but only runs on this CPU)" OFF)
option(GO_LTO    "Link-time optimization in Release builds" ON)

include(CheckIPOSupported)
check_ipo_supported(RESULT IPO_OK OUTPUT IPO_MSG LANGUAGES CXX)
if(GO_LTO AND NOT IPO_OK)
    message(WARNING "LTO not supported by this toolchain: ${IPO_MSG}")
endif()

function(go_optimize target)
    if(GO_NATIVE)
        target_compile_options(${target} PRIVATE -march=native)
    endif()
    if(GO_LTO AND IPO_OK)
        set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    endif()
endfunction()
set (SOURCES
    src/mode/gameSession.cpp
    src/ui/fonts.cpp
    src/ui/hudView.cpp
    src/ui/board.cpp
    src/GameLogic.cpp
    src/ui/audio.cpp
    src/ui/button.cpp
    src/ui/menu.cpp
    src/Game.cpp
    src/GameParser.cpp
)

add_library(GoCore STATIC ${SOURCES})
go_optimize(GoCore)
target_include_directories(GoCore PUBLIC include)
target_link_libraries(GoCore PUBLIC sfml-graphics sfml-window sfml-system sfml-audio)

function(add_go_program e)
    add_executable(${e} src/${e}.cpp)
    target_link_libraries(${e} PRIVATE GoCore)

     # SFML audio needs OpenAL
    add_custom_command(TARGET ${e} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${sfml_SOURCE_DIR}/extlibs/bin/x64/openal32.dll"
            $<TARGET_FILE_DIR:${e}>)

    # winpthread comes from the MinGW toolchain folder (next to g++.exe)
    get_filename_component(MINGW_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
    if(EXISTS "${MINGW_BIN}/libwinpthread-1.dll")
        add_custom_command(TARGET ${e} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${MINGW_BIN}/libwinpthread-1.dll"
                $<TARGET_FILE_DIR:${e}>)
    endif()

    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_link_options(${e} PRIVATE -static-libgcc -static-libstdc++)
    endif()

    add_custom_command(TARGET ${e} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:${e}>/assets)

    if(WIN32)
        add_custom_command(TARGET ${e} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_BINARY_DIR}/_deps/sfml-src/extlibs/bin/x64/openal32.dll"
            $<TARGET_FILE_DIR:${e}>)
    endif()
    go_optimize(${e}) 
endfunction()

add_go_program(main)
add_go_program(ui-dev)
